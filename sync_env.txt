# =====================================================
# 环境同步配置文件（声明式）
# =====================================================

[JAVA]
JAVA_VERSION=17
JAVA_CANDIDATES=temurin@17,zulu@17,openjdk@17

[ANDROID]
ANDROID_SDK_DEFAULT=$HOME/Library/Android/sdk
ANDROID_STUDIO_APP=/Applications/Android Studio.app

[FLUTTER]
# USE_FVM=true 时做两件事：
#   1) 把 FLUTTER_CANDIDATES 里第一个真实存在的目录挂到 PATH（兜底，任何 shell 都能直接用）；
#   2) 在交互式 shell 挂载 flutter()/dart()，版本按当前项目的 .fvmrc 解析，而不是固定用 default。
# FLUTTER_CANDIDATES 逗号分隔，按顺序取第一个存在且为目录的路径。
USE_FVM=true
FLUTTER_CANDIDATES=$HOME/fvm/default/bin,$HOME/development/flutter/bin

[NODE]
ENABLE_PNPM=true
ENABLE_COREPACK=true
NVM_DIR=$HOME/.nvm

[RUST]
ENABLE_CARGO=true
CARGO_HOME=$HOME/.cargo

[PYTHON]
ENABLE_PYENV=true
PYENV_ROOT=$HOME/.pyenv

[RUBY]
ENABLE_RBENV=true
RBENV_ROOT=$HOME/.rbenv

[GO]
ENABLE_GO=true
GOPATH=$HOME/go

[PATH]
$HOME/bin
$HOME/.local/bin
$HOME/Applications/Visual Studio Code.app/Contents/Resources/app/bin
/Applications/Visual Studio Code.app/Contents/Resources/app/bin
$HOME/.pub-cache/bin
$HOME/.cargo/bin
$HOME/go/bin

[ALIASES]
ll=ls -alF
la=ls -A
l=ls -CF
gs=git status
ga=git add .
gc=git commit
gp=git push
gl=git pull
